cmake_minimum_required(VERSION 3.20)
project(UnifiedWorkbench VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  add_compile_options(/utf-8)
  add_compile_definitions(UNICODE _UNICODE)
endif()
enable_testing()
find_package(Threads REQUIRED)
add_library(core STATIC src/core/numerics.cpp src/core/parsers.cpp src/core/batch.cpp src/core/hardware.cpp src/core/jobs.cpp src/core/verification.cpp)
target_link_libraries(core PUBLIC Threads::Threads)
target_include_directories(core PUBLIC src)
add_library(dependencies INTERFACE)
target_include_directories(dependencies SYSTEM INTERFACE third_party)
add_library(platform STATIC src/platform/atomic_file.cpp)
target_link_libraries(platform PUBLIC core)
add_library(server STATIC src/server/server.cpp)
target_link_libraries(server PUBLIC core dependencies)
add_executable(wb_cli src/cli/main.cpp)
target_link_libraries(wb_cli PRIVATE platform dependencies)
add_executable(wb_bench src/cli/benchmark.cpp)
target_link_libraries(wb_bench PRIVATE platform dependencies)
add_executable(generate_samples tools/generate_samples.cpp)
target_link_libraries(generate_samples PRIVATE core)
if(WIN32)
  add_executable(Launch WIN32 Launch/main.cpp)
  target_link_libraries(Launch PRIVATE shell32)
else()
  add_executable(Launch Launch/main.cpp)
endif()
target_link_libraries(Launch PRIVATE server)
add_executable(core_tests tests/numerics.cpp tests/parsers.cpp tests/batch.cpp tests/hardware.cpp tests/jobs.cpp)
target_link_libraries(core_tests PRIVATE core dependencies)
add_executable(server_tests tests/server.cpp)
target_link_libraries(server_tests PRIVATE server)
add_test(NAME server_security COMMAND server_tests)
add_executable(atomic_tests tests/atomic_file.cpp)
target_link_libraries(atomic_tests PRIVATE platform dependencies)
add_test(NAME atomic_exports COMMAND atomic_tests)
if(WIN32)
  target_compile_definitions(server PUBLIC _WIN32_WINNT=0x0A00)
  target_link_libraries(server PUBLIC ws2_32)
endif()
foreach(t platform atomic_tests core server wb_cli wb_bench Launch core_tests server_tests generate_samples)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /WX)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endforeach()
add_test(NAME numerics COMMAND core_tests)
add_test(NAME launcher_scaffold_smoke COMMAND Launch --smoke-test)
file(GLOB_RECURSE CORE_FILES CONFIGURE_DEPENDS src/core/*.cpp src/core/*.hpp)
foreach(f ${CORE_FILES})
  file(READ "${f}" text)
  if(text MATCHES "#include[^\n]*(windows.h|httplib|server/|web/|sys/socket)")
    message(FATAL_ERROR "Core dependency rule violated")
  endif()
endforeach()

if(EXISTS "${CMAKE_SOURCE_DIR}/web/dist/index.html")
  add_custom_command(TARGET Launch POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/web/dist" "$<TARGET_FILE_DIR:Launch>/web")
endif()

add_executable(browser_host tools/browser_host.cpp)
target_link_libraries(browser_host PRIVATE server)
add_executable(challenge tools/challenge.cpp)
target_link_libraries(challenge PRIVATE core dependencies)
